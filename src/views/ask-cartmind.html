SELECT c.customer_id, c.rfm_segment, COUNT(o.order_id) AS weekend_orders, AVG(o.total_amount) AS aov
FROM customers c 
JOIN orders o ON c.customer_id = o.customer_id
WHERE c.rfm_segment = 'At Risk' 
  AND EXTRACT(ISODOW FROM o.order_date) IN (6, 7)
  AND EXISTS (
    SELECT 1 FROM order_items oi 
    JOIN products p ON oi.sku = p.sku 
    WHERE oi.order_id = o.order_id AND p.category = 'Baby Care'
  )
GROUP BY c.customer_id, c.rfm_segment 
HAVING COUNT(o.order_id) >= 3;
